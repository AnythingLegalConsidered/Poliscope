
2. Create `playwright.config.ts` at project root:
   ```typescript
   import { fileURLToPath } from 'node:url'
   import { defineConfig, devices } from '@playwright/test'
   import type { ConfigOptions } from '@nuxt/test-utils/playwright'

   export default defineConfig<ConfigOptions>({
     testDir: './e2e',
     timeout: 60000,
     use: {
       nuxt: {
         rootDir: fileURLToPath(new URL('.', import.meta.url)),
       },
     },
     projects: [
       { name: 'chromium', use: { ...devices['Desktop Chrome'] } },
     ],
   })
   ```

<task type="auto">
  <name>Task 2: E2E spec files for 4 critical journeys</name>
  <files>e2e/home.spec.ts, e2e/debate.spec.ts, e2e/deputies.spec.ts, e2e/search.spec.ts</files>
  <action>
All spec files import `{ expect, test }` from `@nuxt/test-utils/playwright` and use `goto` helper with `{ waitUntil: 'hydration' }`.

Use `data-testid` selectors for component locators: `page.locator('[data-testid="debate-card"]')`.

Important: The app requires a running PostgreSQL database with data. Tests run against a live Nuxt dev server started by @nuxt/test-utils. If the database has no data, tests asserting "at least one card" will fail — this is expected and correct (tests verify real behavior).

1. `e2e/home.spec.ts` — Home page (debates list):
   - Test "home page loads with debate cards": goto `/`, assert h1 heading visible, assert at least one `[data-testid="debate-card"]` is visible
   - Test "infinite scroll loads more debates": goto `/`, scroll to bottom, wait briefly, verify more debate cards appeared (count > initial count)
   - Test "debate card links to debate page": goto `/`, click first debate card link, assert URL matches `/debates/\d+`

2. `e2e/debate.spec.ts` — Debate detail page:
   - Test "debate page renders title and interventions": goto `/debates/1`, assert h1 is not empty, assert at least one `[data-testid="intervention-card"]` visible
   - Test "intervention cards show speaker names": goto `/debates/1`, assert first intervention card contains text (not empty)

3. `e2e/deputies.spec.ts` — Deputies list and detail:
   - Test "deputies page loads with deputy cards": goto `/deputies`, assert at least one `[data-testid="deputy-card"]` visible
   - Test "deputies search filters list": goto `/deputies`, fill search input with a deputy name fragment, wait for list to update, verify filtered results
   - Test "deputy card links to profile": goto `/deputies`, click first deputy card link, assert URL matches `/deputies/\d+`

4. `e2e/search.spec.ts` — Search flow:
   - Test "header search bar navigates to search page": goto `/`, fill `input[type="search"]` with "budget", press Enter, assert URL matches `/search\?q=budget`
   - Test "search results page shows results": goto `/search?q=assemblee`, wait for `[data-testid="search-result-card"]`, assert at least one is visible
   - Test "search with no results shows empty state": goto `/search?q=xyznonexistent123`, assert no result cards visible or empty state message shown

Each test should use Playwright's auto-retry assertions (`await expect(locator).toBeVisible()`) rather than manual waits to avoid flakiness.
