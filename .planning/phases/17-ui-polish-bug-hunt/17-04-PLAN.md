    2. **console-errors.spec.ts** — cross-route error detection
       ```ts
       import { expect, test } from '@nuxt/test-utils/playwright'

       // Note: / IS the debates index — there is no separate /debates route
       const ROUTES = ['/', '/deputies', '/votes', '/search', '/about', '/legal']

       // whitelist known benign warnings/errors — start empty, add only when proven benign
       const WHITELIST: RegExp[] = [
         // e.g. /favicon\.ico/,
       ]

       for (const route of ROUTES) {
         test(`no console errors on ${route}`, async ({ page, goto }) => {
           const errors: string[] = []
           page.on('pageerror', (err) => errors.push(`pageerror: ${err.message}`))
           page.on('console', (msg) => {
             if (msg.type() === 'error') {
               const text = msg.text()
               if (!WHITELIST.some((re) => re.test(text))) {
                 errors.push(`console.error: ${text}`)
               }
             }
           })
           await goto(route, { waitUntil: 'hydration' })
           await page.waitForTimeout(500)
           expect(errors, `Errors on ${route}:\n${errors.join('\n')}`).toHaveLength(0)
         })
       }

       // Detail pages — resolved dynamically from their list page's first card,
       // because / /deputies /votes all render cards with hrefs we can harvest.
       // This is where hydration mismatches and SSR/client divergence are most likely.
       const detailRoutes = async (page, goto): Promise<string[]> => {
         const results: string[] = []
         for (const [listPath, cardTestId] of [
           ['/', 'debate-card'],          // / IS the debates index
           ['/deputies', 'deputy-card'],
           ['/votes', 'vote-card'],
         ] as const) {
           await goto(listPath, { waitUntil: 'hydration' })
           const href = await page
             .locator(`[data-testid="${cardTestId}"] a`)
             .first()
             .getAttribute('href')
           if (href) results.push(href)
         }
         return results
       }

       test('no console errors on detail pages (resolved from list pages)', async ({ page, goto }) => {
         const urls = await detailRoutes(page, goto)
         expect(urls.length).toBeGreaterThan(0)
         for (const url of urls) {
           const errors: string[] = []
           page.on('pageerror', (err) => errors.push(`pageerror on ${url}: ${err.message}`))
           page.on('console', (msg) => {
             if (msg.type() === 'error') {
               const text = msg.text()
               if (!WHITELIST.some((re) => re.test(text))) {
                 errors.push(`console.error on ${url}: ${text}`)
               }
             }
           })
           await goto(url, { waitUntil: 'hydration' })
           await page.waitForTimeout(500)
           expect(errors, `Errors on ${url}:\n${errors.join('\n')}`).toHaveLength(0)
         }
       })
       ```
